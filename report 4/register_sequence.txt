    autonumber off
    actor User as User
    participant UI as :RegisterUI
    participant Controller as :AuthController
    participant Service as :AuthService
    participant UserModel as :User
    participant DB as :MongoDB

    User->>UI: 1: Enter registration details (fullName, phone, email, password, role)
    User->>UI: 1.1: Click "Register"
    UI->>Controller: 1.2: register(registerDto)
    Controller->>Service: 1.2.1: register(userData)

    Service->>UserModel: 1.2.1.1: findOne({ $or: [{ phone }, { email }] })
    UserModel->>DB: 1.2.1.1.1: db.users.findOne(...)
    DB-->>UserModel: return existingUser
    UserModel-->>Service: return existingUser

    alt [Phone or Email Already Registered]
        Service-->>Controller: throw DuplicateResourceException
        Controller-->>UI: return 409 Conflict
        UI-->>User: Display notification "Phone number or Email already exists"
    else [Account Details Available (Happy Path)]
        Service->>Service: 1.2.1.2: hashPassword(password)
        Service->>UserModel: 1.2.1.3: create(newUserData)
        UserModel->>DB: 1.2.1.3.1: db.users.insertOne({ phone, email, fullName, passwordHash, roles, status })
        DB-->>UserModel: return createdUserDocument
        UserModel-->>Service: return createdUser

        Service->>Service: 1.2.1.4: generateTokens(createdUser)
        Service->>UserModel: 1.2.1.5: updateOne({ _id: userId }, { $push: { sessions: sessionData } })
        UserModel->>DB: 1.2.1.5.1: db.users.updateOne({ $push: { sessions } })
        DB-->>UserModel: confirm session saved

        Service-->>Controller: return authResponseDto (user, tokens)
        Controller-->>UI: return 201 Created (authResponseDto)
        UI->>UI: 1.3: Save tokens into LocalStorage/AsyncStorage
        UI-->>User: 1.4: Display "Registration successful" & Redirect to Home Dashboard
    end
