    %% 1. TẦNG VIEW (WEB DASHBOARD)
    class AuditLogsView {
        +auditLogs: AuditLogItem[]
        +auditRoleFilter: string
        +auditSearchTerm: string
        +loading: boolean
        +loadLogs()
        +filterByRole(role)
        +filterBySearchTerm(term)
        +formatAuditDateTime(dateStr)
        +formatRoleBadge(roleStr)
        +extractActionDetails(metadata)
    }

    %% 2. TẦNG CONTROLLER
    class AuditController {
        +getAuditLogs(req, res, next)
        +getAuditLogById(req, res, next)
    }

    %% 3. TẦNG MIDDLEWARE
    class AuthMiddleware {
        +authenticate(req, res, next)
    }

    class RoleMiddleware {
        +authorize(roles)(req, res, next)
    }

    %% 4. TẦNG SERVICE
    class AuditService {
        +getAuditLogs(query, currentUser): Promise
        +getAuditLogById(id): Promise
        +getDescendantAreaIds(rootAreaId): Promise
        +log(params): Promise
    }

    %% 5. TẦNG MONGOOSE MODELS / SCHEMAS (KHÔNG DÙNG REPOSITORY TRUNG GIAN)
    class AuditLog {
        +actorId: ObjectId
        +actorRole: string
        +action: string
        +entityType: string
        +entityId: ObjectId
        +areaId: ObjectId
        +metadata: Mixed
        +createdAt: Date
    }

    class User {
        +userId: ObjectId
        +fullName: string
        +phone: string
        +email: string
        +roles: Array
        +authority: Object
    }

    class AdministrativeArea {
        +areaId: ObjectId
        +name: string
        +code: string
        +type: string
        +parentId: ObjectId
        +status: string
    }

    %% ================= QUAN HỆ KIẾN TRÚC NODE.JS =================
    %% View gọi API
    AuditLogsView ..> AuditController : HTTP GET /audit-logs

    %% Middleware xác thực & phân quyền
    AuditController ..> AuthMiddleware : authenticates
    AuditController ..> RoleMiddleware : authorizes ADMIN, LOCAL_AUTHORITY

    %% Controller gọi Service
    AuditController ..> AuditService : uses

    %% Service thao tác trực tiếp với Mongoose Models
    AuditService ..> AuditLog : queries & populates
    AuditService ..> User : populates actorId
    AuditService ..> AdministrativeArea : populates areaId & filters descendants

    %% Quan hệ giữa các Models trong Database
    AuditLog "0..*" -- "1" User : performed by
    AuditLog "0..*" -- "0..1" AdministrativeArea : scoped to
    AdministrativeArea "0..*" -- "0..1" AdministrativeArea : child of (parentId)
