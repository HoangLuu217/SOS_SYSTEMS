    %% 1. TẦNG CONTROLLER
    class ChatController {
        +getConversations()
        +getMessages(channel, limit)
        +sendMessage(messageDto)
        +markAsRead(markDto)
        +streamChatEvents()
    }

    %% 2. TẦNG SERVICE
    class ChatService {
        +getConversations(currentUser)
        +getMessages(filterDto)
        +sendMessage(senderUser, data)
        +markAsRead(currentUser, data)
        +emitRealtimeMessage(channel, msg)
    }

    %% 3. TẦNG MONGOOSE MODELS / SCHEMAS (KHÔNG DÙNG REPOSITORY TRUNG GIAN)
    class ChatMessage {
        +messageId
        +senderId
        +senderName
        +senderRole
        +recipientId
        +channel
        +content
        +attachments
        +readBy
        +createdAt
    }

    class User {
        +userId
        +fullName
        +phone
        +roles
        +authority
        +avatarUrl
    }

    %% 4. CỤM TỔ CHỨC / ĐỘI CỨU HỘ
    class AuthorityOrganization {
        +orgId
        +name
        +code
        +type
    }

    class RescueTeam {
        +teamId
        +teamName
        +leaderId
        +status
    }

    %% ================= QUAN HỆ KIẾN TRÚC NODE.JS =================
    %% Controller gọi Service
    ChatController ..> ChatService : uses

    %% Service thao tác trực tiếp với Mongoose Models
    ChatService ..> ChatMessage : manages
    ChatService ..> User : queries
