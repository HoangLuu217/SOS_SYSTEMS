sequenceDiagram
    autonumber off
    actor User as User
    participant UI as :EditProfileUI
    participant Controller as :UserController
    participant Service as :UserService
    participant UserModel as :User
    participant Cloud as :CloudinaryService
    participant DB as :MongoDB

    User->>UI: 1: Update profile fields (fullName, dob, gender, address, phone, email) & select new avatar
    User->>UI: 1.1: Click "Save Changes"

    opt [New Avatar Uploaded]
        UI->>Controller: 1.2: updateAvatar(file)
        Controller->>Service: 1.2.1: updateAvatar(userId, file)
        Service->>Cloud: 1.2.1.1: upload(imageStream)
        Cloud-->>Service: return secure_url
        Service->>UserModel: 1.2.1.2: updateOne({ _id: userId }, { avatarUrl: secure_url })
        UserModel->>DB: 1.2.1.2.1: db.users.updateOne({ _id: userId }, { $set: { avatarUrl } })
        DB-->>UserModel: confirm updated
        Service-->>Controller: return new avatarUrl
        Controller-->>UI: return 200 OK (avatar updated)
    end

    UI->>Controller: 1.3: updateMe(updateProfileDto)
    Controller->>Service: 1.3.1: updateMe(userId, updateData)

    Service->>Service: 1.3.1.1: validateProfileData(updateData)

    opt [Phone or Email Modified]
        Service->>UserModel: 1.3.1.2: findOne({ $or: [{ phone }, { email }], _id: { $ne: userId } })
        UserModel->>DB: 1.3.1.2.1: db.users.findOne(...)
        DB-->>UserModel: return existingDuplicate
        UserModel-->>Service: return existingDuplicate
    end

    alt [Duplicate Phone or Email Found]
        Service-->>Controller: throw DuplicateResourceException
        Controller-->>UI: return 409 Conflict
        UI-->>User: Display notification "Phone number or Email is already in use"
    else [Valid Information (Happy Path)]
        Service->>UserModel: 1.3.1.3: updateOne({ _id: userId }, updateData)
        UserModel->>DB: 1.3.1.3.1: db.users.updateOne({ _id: userId }, { $set: updateData })
        DB-->>UserModel: confirm update
        UserModel-->>Service: return updatedUser

        Service-->>Controller: return updatedProfileDto
        Controller-->>UI: return 200 OK (Profile updated)
        UI->>UI: 1.4: Update local cached profile state
        UI-->>User: 1.5: Display notification "Profile updated successfully" & return to Profile Screen
    end
