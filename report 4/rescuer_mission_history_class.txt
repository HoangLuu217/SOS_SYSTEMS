    class VehicleRepository {
        <<interface>>
        +findAllAvailable()

        +updateStatus(id, status)
    }

    class SosAssignmentRepository {
        <<interface>>
        +save(assignment)
        +findById(id)
        +findBySosRequest(sosId)
    }

    class SosRequestRepository {
        <<interface>>
        +save(sosRequest)
        +findById(id)
        +updateStatus(id, status)
    }

    class Vehicle {

        +code
        +type
        +capacity
        +status
    }

    class SosAssignment {
        +assignmentId
        +missionType
        +status
        +priority
        +assignedAt
    }

    class SosRequest {
        +sosRequestId
        +code
        +severity
        +victimCount
        +status
    }

    class AuthorityOrganization {
        +orgId
        +name
        +code
        +type
    }

    class RescueTeam {

        +teamName
        +leaderId
        +status
    }

    %% ================= QUAN HỆ & LIÊN KẾT =================
    %% Controller -> Service
    SosAssignmentController ..> SosAssignmentService : uses

    %% Service -> Repositories
    SosAssignmentService ..> VehicleRepository
    SosAssignmentService ..> SosAssignmentRepository
    SosAssignmentService ..> SosRequestRepository

    %% Repositories -> Entities
    VehicleRepository ..> Vehicle : manages
    SosAssignmentRepository ..> SosAssignment : manages
    SosRequestRepository ..> SosRequest : manages

    %% Entity relationships
    SosAssignment "1" --> "0..*" Vehicle
    SosRequest "1" *-- "1..*" SosAssignment

    %% Right column: Organization -> Team -> SosRequest
    AuthorityOrganization ..> RescueTeam : manages
    RescueTeam "1" --> "1" SosRequest
