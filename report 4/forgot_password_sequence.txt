sequenceDiagram
    autonumber off
    actor User as User
    participant UI as :ForgotPasswordUI
    participant Controller as :AuthController
    participant Service as :AuthService
    participant UserModel as :User
    participant DB as :MongoDB

    %% PHASE 1: REQUEST PASSWORD RESET OTP
    User->>UI: 1: Enter registered phone/email
    User->>UI: 1.1: Click "Send verification code"
    UI->>Controller: 1.2: forgotPassword(forgotDto)
    Controller->>Service: 1.2.1: forgotPassword(identifier)

    Service->>UserModel: 1.2.1.1: findOne({ $or: [{ phone }, { email }] })
    UserModel->>DB: 1.2.1.1.1: db.users.findOne(...)
    DB-->>UserModel: return user
    UserModel-->>Service: return user

    alt [User Not Found]
        Service-->>Controller: throw NotFoundException
        Controller-->>UI: return 404 Not Found
        UI-->>User: Display notification "Account not found"
    else [User Found]
        Service->>Service: 1.2.1.2: generateOtpToken()
        Service->>Service: 1.2.1.3: hashToken(token)
        Service->>UserModel: 1.2.1.4: updateOne({ _id: userId }, { resetPasswordTokenHash, resetPasswordExpires })
        UserModel->>DB: 1.2.1.4.1: db.users.updateOne(...)
        DB-->>UserModel: confirm update
        Service->>Service: 1.2.1.5: sendOtpNotification(phone/email, otp)
        Service-->>Controller: return success message
        Controller-->>UI: return 200 OK (OTP sent)
        UI-->>User: 1.3: Display notification "Verification code sent to your phone/email"

        %% PHASE 2: VERIFY OTP AND RESET PASSWORD
        User->>UI: 2: Enter received OTP and new password
        User->>UI: 2.1: Click "Reset Password"
        UI->>Controller: 2.2: resetPassword(resetDto)
        Controller->>Service: 2.2.1: resetPassword(identifier, token, newPassword)

        Service->>UserModel: 2.2.1.1: findOne({ $or: [{ phone }, { email }] })
        UserModel->>DB: 2.2.1.1.1: db.users.findOne(...)
        DB-->>UserModel: return user
        Service->>Service: 2.2.1.2: verifyOtpHash(token, user.resetPasswordTokenHash, user.resetPasswordExpires)

        alt [Invalid or Expired OTP]
            Service-->>Controller: throw BadRequestException
            Controller-->>UI: return 400 Bad Request
            UI-->>User: Display notification "Invalid or expired OTP code"
        else [Valid OTP (Happy Path)]
            Service->>Service: 2.2.1.3: hashPassword(newPassword)
            Service->>UserModel: 2.2.1.4: updateOne({ _id: userId }, { passwordHash, resetPasswordTokenHash: null, sessions: [] })
            UserModel->>DB: 1.2.1.4.1: db.users.updateOne({ $set: { passwordHash }, $set: { sessions: [] } })
            DB-->>UserModel: confirm update

            Service-->>Controller: return success message
            Controller-->>UI: return 200 OK
            UI-->>User: 2.3: Display "Password reset successfully" & Redirect to Login
        end
    end
