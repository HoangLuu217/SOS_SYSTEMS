    %% 1. TẦNG VIEW (WEB DASHBOARD)
    class ReportsView {
        +stats: DashboardSummaryStats
        +sosList: SosRequestItem[]
        +auditLogs: AuditLogItem[]
        +timeRangeLabel: string
        +scopeName: string
        +handleExportSummaryCsv()
        +handleExportExcel()
    }

    class AuditLogsView {
        +auditLogs: AuditLogItem[]
        +handleExportAuditCsv()
    }

    %% 2. TẦNG EXPORT UTILITY (CLIENT-SIDE)
    class ExportReportsExcel {
        +exportDashboardWorkbook(rawData: ReportsRawData): void
        +buildAllSheets(rawData): SheetSpec[]
        +buildWorksheetXml(sheet): string
        +buildStylesXml(): string
        +createZipArchive(files): Uint8Array
    }

    class CsvExporter {
        +escapeCsvCell(val): string
        +buildCsvContent(headers, rows): string
        +triggerBlobDownload(blob, filename): void
    }

    %% 3. TẦNG CONTROLLER
    class AuditController {
        +getAuditLogs(req, res, next)
        +getAuditLogById(req, res, next)
    }

    class SosController {
        +getSosRequests(req, res, next)
    }

    class AdminController {
        +getStatistics(req, res, next)
    }

    %% 4. TẦNG SERVICE
    class AuditService {
        +getAuditLogs(query, currentUser): Promise
        +getDescendantAreaIds(rootAreaId): Promise
    }

    class SosService {
        +getSosRequests(query, user): Promise
    }

    %% 5. TẦNG MONGOOSE MODELS / SCHEMAS (KHÔNG DÙNG REPOSITORY TRUNG GIAN)
    class AuditLog {
        +actorId: ObjectId
        +actorRole: string
        +action: string
        +entityType: string
        +entityId: ObjectId
        +areaId: ObjectId
        +metadata: Mixed
        +createdAt: Date
    }

    class SosRequest {
        +sosCode: string
        +citizenId: ObjectId
        +emergencyType: string
        +priority: string
        +people: number
        +address: string
        +status: string
        +createdAt: Date
    }

    class User {
        +userId: ObjectId
        +fullName: string
        +phone: string
        +email: string
        +roles: Array
    }

    %% ================= QUAN HỆ KIẾN TRÚC =================
    %% View gọi Export Utilities
    ReportsView ..> ExportReportsExcel : calls exportDashboardWorkbook()
    ReportsView ..> CsvExporter : calls for CSV export
    AuditLogsView ..> CsvExporter : calls for CSV export

    %% View gọi API Controllers
    ReportsView ..> AuditController : fetches audit data
    ReportsView ..> SosController : fetches SOS data
    ReportsView ..> AdminController : fetches statistics

    %% Controller gọi Service
    AuditController ..> AuditService : uses
    SosController ..> SosService : uses

    %% Service thao tác trực tiếp với Mongoose Models
    AuditService ..> AuditLog : queries
    SosService ..> SosRequest : queries
    AuditService ..> User : populates

    %% Quan hệ giữa các Models trong Database
    AuditLog "0..*" -- "1" User : performed by
    SosRequest "0..*" -- "1" User : reported by
