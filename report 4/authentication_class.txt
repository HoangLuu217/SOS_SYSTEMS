classDiagram
    direction TB
    classDef default fill:#ffffff,stroke:#000000,stroke-width:1px,color:#000000;

    %% 1. TẦNG CONTROLLER
    class AuthController {
        +register(registerDto)
        +login(loginDto)
        +loginWithGoogle(googleDto)
        +logout(refreshToken)
        +forgotPassword(forgotDto)
        +resetPassword(resetDto)
        +changePassword(changePasswordDto)
    }

    %% 2. TẦNG SERVICE
    class AuthService {
        +register(userData)
        +login(credentials)
        +loginWithGoogle(googleData)
        +logout(userId, refreshToken)
        +forgotPassword(identifier)
        +resetPassword(resetData)
        +changePassword(userId, passwordDto)
        +generateTokens(user)
    }

    %% 3. TẦNG MONGOOSE MODELS / SCHEMAS (KHÔNG DÙNG REPOSITORY TRUNG GIAN)
    class UserSession {
        +sessionId
        +refreshTokenHash
        +userAgent
        +ipAddress
        +expiresAt
        +createdAt
    }

    class User {
        +userId
        +phone
        +email
        +fullName
        +passwordHash
        +googleId
        +roles
        +status
        +resetPasswordTokenHash
        +resetPasswordExpires
        +sessions
        +comparePassword(password)
        +toSafeObject()
    }

    class Role {
        +roleCode
        +roleName
        +description
    }

    %% 4. CỤM TỔ CHỨC / ĐỘI CỨU HỘ
    class AuthorityOrganization {
        +orgId
        +name
        +code
        +type
        +phone
        +email
    }

    class RescueTeam {
        +teamId
        +teamName
        +leaderId
        +status
    }

    %% ================= QUAN HỆ KIẾN TRÚC NODE.JS =================
    %% Controller gọi Service
    AuthController ..> AuthService : uses

    %% Service thao tác trực tiếp với Mongoose Models
    AuthService ..> User : manages
    AuthService ..> UserSession : manages

    %% Quan hệ giữa các Models trong Database
    User "1" *-- "0..*" UserSession : contains
    User "1..*" -- "1..*" Role : has

    AuthorityOrganization ..> RescueTeam : manages
    RescueTeam "1" -- "1" User : led by
