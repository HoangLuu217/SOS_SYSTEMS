sequenceDiagram
    autonumber off
    actor User as User
    participant UI as :LoginUI
    participant Controller as :AuthController
    participant Service as :AuthService
    participant UserModel as :User
    participant DB as :MongoDB

    User->>UI: 1: Enter phone/email & password
    User->>UI: 1.1: Click "Login"
    UI->>Controller: 1.2: login(requestDto)
    Controller->>Service: 1.2.1: login(loginDto)
    
    Service->>UserModel: 1.2.1.1: findOne({ $or: [{ phone }, { email }] })
    UserModel->>DB: 1.2.1.1.1: db.users.findOne(...)
    DB-->>UserModel: return userDocument
    UserModel-->>Service: return user

    alt [User Not Found]
        Service-->>Controller: throw NotFoundException / Unauthorized
        Controller-->>UI: return 401 Unauthorized
        UI-->>User: Display notification "Account not found"
    else [User Found]
        Service->>Service: 1.2.1.2: comparePassword(password, user.passwordHash)

        alt [Invalid Password]
            Service-->>Controller: throw InvalidPasswordException
            Controller-->>UI: return 401 Unauthorized
            UI-->>User: Display notification "Incorrect password"
        else [Valid Password (Happy Path)]
            opt [Google Account without password]
                Service->>UserModel: 1.2.1.3: updateOne({ _id: userId }, { passwordHash })
                UserModel->>DB: 1.2.1.3.1: db.users.updateOne(...)
                DB-->>UserModel: confirm updated
            end

            Service->>Service: 1.2.1.4: generateTokens(user)
            Service->>UserModel: 1.2.1.5: updateOne({ _id: userId }, { $push: { sessions: sessionData } })
            UserModel->>DB: 1.2.1.5.1: db.users.updateOne({ $push: { sessions } })
            DB-->>UserModel: confirm session saved

            Service-->>Controller: return authResponseDto (user, tokens)
            Controller-->>UI: return 200 OK (authResponseDto)
            UI->>UI: 1.3: Save tokens into LocalStorage/AsyncStorage
            UI-->>User: 1.4: Redirect to Home Dashboard
        end
    end
