    autonumber off
    actor User as User
    participant UI as :ChangePasswordUI
    participant Controller as :AuthController
    participant Service as :UserService
    participant UserModel as :User
    participant DB as :MongoDB

    User->>UI: 1: Enter current password, new password & confirm password
    User->>UI: 1.1: Click "Update Password"
    UI->>Controller: 1.2: changePassword(changePasswordDto)
    Controller->>Service: 1.2.1: changePassword(userId, passwordDto)

    Service->>UserModel: 1.2.1.1: findById(userId)
    UserModel->>DB: 1.2.1.1.1: db.users.findOne({ _id: userId })
    DB-->>UserModel: return userDocument
    UserModel-->>Service: return user

    alt [User Not Found or Unauthorized]
        Service-->>Controller: throw UnauthorizedException
        Controller-->>UI: return 401 Unauthorized
        UI-->>User: Display notification "Please log in again"
    else [User Authenticated]
        Service->>Service: 1.2.1.2: comparePassword(oldPassword, user.passwordHash)

        alt [Current Password Incorrect]
            Service-->>Controller: throw InvalidPasswordException
            Controller-->>UI: return 400 Bad Request
            UI-->>User: Display notification "Current password is not correct"
        else [Current Password Valid (Happy Path)]
            Service->>Service: 1.2.1.3: hashPassword(newPassword)
            Service->>UserModel: 1.2.1.4: updateOne({ _id: userId }, { passwordHash: newPasswordHash })
            UserModel->>DB: 1.2.1.4.1: db.users.updateOne({ _id: userId }, { $set: { passwordHash } })
            DB-->>UserModel: confirm updated
            UserModel-->>Service: confirm password updated

            Service-->>Controller: return success message
            Controller-->>UI: return 200 OK (Password changed)
            UI-->>User: 1.3: Display notification "Password changed successfully"
        end
    end
